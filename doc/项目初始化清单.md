## 7.2 `packages/domain/src/merge.ts`

```ts
import type { GraphEdge, GraphNode, MergedGraph, SegmentGraph } from '@listen-graphic/schema';
import { normalizeText } from './normalize';

function canMergeNode(a: GraphNode, b: GraphNode): boolean {
  if (a.type !== b.type) return false;
  if (a.speaker && b.speaker && a.speaker !== b.speaker) return false;

  return normalizeText(a.text) === normalizeText(b.text);
}

function mergeNodes(prev: GraphNode[], next: GraphNode[]): GraphNode[] {
  const result = [...prev];

  for (const node of next) {
    const existing = result.find((n) => canMergeNode(n, node));
    if (existing) {
      existing.confidence = Math.max(existing.confidence ?? 0, node.confidence ?? 0);
      existing.evidenceChunkIds = Array.from(
        new Set([...(existing.evidenceChunkIds ?? []), ...(node.evidenceChunkIds ?? [])]),
      );
    } else {
      result.push(node);
    }
  }

  return result;
}

function mergeEdges(prev: GraphEdge[], next: GraphEdge[]): GraphEdge[] {
  const result = [...prev];

  for (const edge of next) {
    const existing = result.find(
      (e) => e.source === edge.source && e.target === edge.target && e.type === edge.type,
    );

    if (existing) {
      existing.confidence = Math.max(existing.confidence ?? 0, edge.confidence ?? 0);
      existing.evidenceChunkIds = Array.from(
        new Set([...(existing.evidenceChunkIds ?? []), ...(edge.evidenceChunkIds ?? [])]),
      );
    } else {
      result.push(edge);
    }
  }

  return result;
}

export function mergeSegmentGraph(prev: MergedGraph, next: SegmentGraph): MergedGraph {
  return {
    ...prev,
    revision: prev.revision + 1,
    nodes: mergeNodes(prev.nodes, next.nodes),
    edges: mergeEdges(prev.edges, next.edges),
    updatedAt: Date.now(),
  };
}
```

import type { GraphNode, MergedGraph } from '@listen-graphic/schema';

function stabilizeNode(node: GraphNode): GraphNode {
  if (node.status === 'confirmed') return node;

  if ((node.confidence ?? 0) >= 0.85) {
    return { ...node, status: 'confirmed' };
  }

  return node;
}

export function stabilizeGraph(graph: MergedGraph): MergedGraph {
  return {
    ...graph,
    nodes: graph.nodes.map(stabilizeNode),
    updatedAt: Date.now(),
  };
}

## 7.4 `packages/domain/src/view-selection.ts`

```ts
import type { MergedGraph, ViewType } from '@listen-graphic/schema';

export function scoreViews(graph: MergedGraph): Record<ViewType, number> {
  const scores: Record<ViewType, number> = {
    timeline: 0,
    causal: 0,
    comparison: 0,
    flow: 0,
    hierarchy: 0,
  };

  for (const edge of graph.edges) {
    if (edge.type === 'follows') {
      scores.timeline += 2;
      scores.flow += 2;
    }
    if (edge.type === 'causes') {
      scores.causal += 3;
    }
    if (edge.type === 'opposes') {
      scores.comparison += 3;
    }
    if (edge.type === 'supports') {
      scores.causal += 1;
      scores.hierarchy += 1;
    }
    if (edge.type === 'example_of') {
      scores.hierarchy += 2;
    }
  }

  return scores;
}

export function pickDefaultView(graph: MergedGraph): ViewType {
  const scores = scoreViews(graph);
  return Object.entries(scores).sort((a, b) => b[1] - a[1])[0][0] as ViewType;
}
```

import type { MergedGraph, SegmentGraph, TranscriptChunk } from '@listen-graphic/schema';

export interface SessionRecord {
  sessionId: string;
  sourceType: 'tab' | 'file';
  createdAt: number;
  endedAt?: number;
  title?: string;
}

export class ListenGraphicDB extends Dexie {
  sessions!: Table<SessionRecord, string>;
  transcriptChunks!: Table<TranscriptChunk, string>;
  segmentGraphs!: Table<SegmentGraph, string>;
  mergedGraphs!: Table<MergedGraph, string>;

  constructor() {
    super('listen-graphic-db');

    this.version(1).stores({
      sessions: 'sessionId, createdAt',
      transcriptChunks: 'chunkId, sessionId, startMs, endMs',
      segmentGraphs: 'segmentId, sessionId, startMs, endMs',
      mergedGraphs: 'sessionId, revision, updatedAt',
    });
  }
}

export const db = new ListenGraphicDB();

import { useMemo } from 'react';
import { ReactFlow, Background, Controls, type Node, type Edge } from 'reactflow';
import 'reactflow/dist/style.css';

import { mockMergedGraph } from './mock-session';
import { toCausalRendererGraph } from '@listen-graphic/graph-renderers';

export default function App() {
  const graph = useMemo(() => toCausalRendererGraph(mockMergedGraph), []);

  const nodes: Node[] = graph.nodes.map((n, index) => ({
    id: n.id,
    position: { x: 100, y: index * 120 },
    data: { label: n.label },
    type: 'default',
  }));

  const edges: Edge[] = graph.edges.map((e) => ({
    id: e.id,
    source: e.source,
    target: e.target,
    label: e.relation,
  }));

  return (
    <div style={{ width: '100vw', height: '100vh' }}>
      <ReactFlow nodes={nodes} edges={edges} fitView>
        <Background />
        <Controls />
      </ReactFlow>
    </div>
  );
}

# 14. 你现在最容易踩的坑，我帮你提前卡掉

## 1）不要一开始就急着装一堆 AI SDK

先 mock provider。
你现在要先证明：

* schema 合理
* graph 可渲染
* merge/stabilize 跑得通

不是先证明“我能调 API”。

## 2）不要先做 extension 再做 web-demo

先把 Web Demo 跑出来，调图和 domain 逻辑舒服很多。
插件环境调试更烦，不适合一上来就拿它做主战场。

## 3）不要第一天就上 Tailwind / UI 库大战

你这个阶段 UI 不是重点。
先把图、会话、消息流立住。

## 4）不要过早拆太多 renderers

先一个 causal renderer，最多再加一个 comparison。
能跑比“设计很完整”更重要。
